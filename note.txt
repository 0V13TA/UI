
		{
			// ROOT CONTAINER (Horizontal Row)
			renderer.element_open(
				ui_ctx,
				{
					style = {
						direction = .ROW,
						width     = lc.ViewPercent{100},
						height    = lc.ViewPercent{100},
						bg_color  = renderer.Color{0.96, 0.96, 0.98, 1}, // Very soft grey background
					},
				},
			)
			defer renderer.element_close(ui_ctx)

			// --- 1. SIDEBAR ---
			{
				renderer.element_open(
					ui_ctx,
					{
						style = {
							direction = .COLUMN,
							width = lc.Fixed{240},
							height = lc.Percent{100},
							bg_color = renderer.Color{1, 1, 1, 1},
							border = [4]f32{0, 1, 0, 0},
							border_color = renderer.Color{0.85, 0.85, 0.85, 1},
							padding = [4]f32{24, 16, 24, 16},
							gap = 24,
						},
					},
				)
				defer renderer.element_close(ui_ctx)

				renderer.text(
					ui_ctx,
					&ev_ctx,
					"OVIETA OS",
					user_style = {font_size = 20, text_color = renderer.Color{0.1, 0.1, 0.1, 1}},
				)

				// Reusing list_view for our sidebar navigation
				renderer.list_view(
					ui_ctx,
					&ev_ctx,
					list_items,
					&list_selected,
					wrapper_style = {
						direction = .COLUMN,
						width = lc.Percent{100},
						height = lc.Grow{1},
						bg_color = renderer.Color{0, 0, 0, 0},
						border = [4]f32{0, 0, 0, 0},
					},
				)
			}

			// --- 2. MAIN CONTENT AREA ---
			{
				renderer.element_open(
					ui_ctx,
					{
						style = {
							direction  = .COLUMN,
							width      = lc.Grow{1},
							height     = lc.Percent{100},
							padding    = [4]f32{32, 48, 32, 48},
							gap        = 32,
							overflow_y = .SCROLL, // Let the main page scroll if things get tall
						},
					},
				)
				defer renderer.element_close(ui_ctx)

				// Header
				renderer.text(
					ui_ctx,
					&ev_ctx,
					list_items[list_selected],
					user_style = {font_size = 32, text_color = renderer.Color{0.1, 0.1, 0.1, 1}},
				)

				// Render distinct pages based on sidebar selection
				if list_selected == 0 {
					// ---- DASHBOARD PAGE ----
					renderer.tabs(ui_ctx, &ev_ctx, tab_labels, &active_tab)

					if active_tab == 0 {
						// Grid-like Form Row
						renderer.element_open(
							ui_ctx,
							{style = {direction = .ROW, gap = 24, width = lc.Percent{100}}},
						)
						defer renderer.element_close(ui_ctx)

						// Column 1
						renderer.element_open(
							ui_ctx,
							{style = {direction = .COLUMN, gap = 16, width = lc.Grow{1}}},
						)
						renderer.dropdown(
							ui_ctx,
							&ev_ctx,
							"Select a role...",
							my_options,
							&my_selected_idx,
							&my_dropdown_open,
							wrapper_style = {
								width = lc.Percent{100},
								border = [4]f32{1, 1, 1, 1},
								padding = [4]f32{12, 16, 12, 16},
								border_radius = [4]f32{6, 6, 6, 6},
								bg_color = renderer.Color{1, 1, 1, 1},
							},
						)
						renderer.combobox(
							ui_ctx,
							&ev_ctx,
							"Search roles...",
							my_options,
							&my_combo_buf,
							&my_combo_idx,
							&my_combo_open,
							wrapper_style = {
								width = lc.Percent{100},
								height = lc.Fixed{50},
								padding = [4]f32{8, 8, 8, 8},
								border = [4]f32{1, 1, 1, 1},
								border_radius = [4]f32{6, 6, 6, 6},
								bg_color = renderer.Color{1, 1, 1, 1},
							},
						)
						renderer.element_close(ui_ctx)

						// Column 2
						renderer.element_open(
							ui_ctx,
							{style = {direction = .COLUMN, gap = 16, width = lc.Grow{1}}},
						)
						renderer.multi_select(
							ui_ctx,
							&ev_ctx,
							"Permissions",
							my_options,
							my_multi_states[:],
							&my_multi_open,
							wrapper_style = {
								width = lc.Percent{100},
								border = [4]f32{1, 1, 1, 1},
								padding = [4]f32{12, 16, 12, 16},
								border_radius = [4]f32{6, 6, 6, 6},
								bg_color = renderer.Color{1, 1, 1, 1},
							},
						)
						renderer.switch_toggle(ui_ctx, &ev_ctx, "Dark Mode", &my_switch_val)
						renderer.element_close(ui_ctx)

						// Card for Sliders and Progress
						renderer.element_open(
							ui_ctx,
							{
								style = {
									direction = .COLUMN,
									gap = 24,
									width = lc.Percent{100},
									padding = [4]f32{24, 24, 24, 24},
									bg_color = renderer.Color{1, 1, 1, 1},
									border = [4]f32{1, 1, 1, 1},
									border_color = renderer.Color{0.9, 0.9, 0.9, 1},
									border_radius = [4]f32{8, 8, 8, 8},
								},
							},
						)

						renderer.text(
							ui_ctx,
							&ev_ctx,
							"Volume Control",
							user_style = {
								font_size = 16,
								text_color = renderer.Color{0.4, 0.4, 0.4, 1},
							},
						)
						renderer.slider(
							ui_ctx,
							&ev_ctx,
							&anim_ctx,
							&my_progress,
							0.0,
							1.0,
							wrapper_style = {
								width = lc.Percent{100},
								height = lc.Fixed{20},
								justify_content = .START,
								align_items = .CENTER,
							},
						)
						renderer.progress_bar(
							ui_ctx,
							my_progress,
							wrapper_style = {
								width = lc.Percent{100},
								height = lc.Fixed{8},
								bg_color = renderer.Color{0.9, 0.9, 0.9, 1},
								border_radius = [4]f32{4, 4, 4, 4},
							},
						)

						renderer.element_open(
							ui_ctx,
							{
								style = {
									direction = .ROW,
									justify_content = .SPACE_BETWEEN,
									align_items = .CENTER,
									width = lc.Percent{100},
								},
							},
						)
						if renderer.button(ui_ctx, &ev_ctx, "Show Toast Overlay") do my_toast_open = true
						renderer.spinner(ui_ctx)
						renderer.element_close(ui_ctx)

						renderer.element_close(ui_ctx) // Close Card

					} else if active_tab == 1 {
						renderer.text(ui_ctx, &ev_ctx, "Statistics content goes here.")
					}

				} else if list_selected == 1 {
					// ---- USER DIRECTORY PAGE ----
					renderer.table(ui_ctx, &ev_ctx, table_headers, table_data, table_cols)

					renderer.element_open(
						ui_ctx,
						{
							style = {
								direction = .ROW,
								justify_content = .END,
								width = lc.Percent{100},
								padding = [4]f32{16, 0, 0, 0},
							},
						},
					)
					if renderer.button(ui_ctx, &ev_ctx, "Delete User", user_style = {bg_color = renderer.Color{0.9, 0.2, 0.2, 1}}) do my_modal_open = true
					renderer.element_close(ui_ctx)

				} else if list_selected == 2 {
					// ---- SETTINGS PAGE ----
					if renderer.accordion_begin(
						ui_ctx,
						&ev_ctx,
						"Advanced Settings",
						&accordion_open,
					) {
						defer renderer.accordion_end(ui_ctx, accordion_open)

						renderer.text(
							ui_ctx,
							&ev_ctx,
							"Warning: Modifying these values may break the layout engine.",
							user_style = {text_color = renderer.Color{0.6, 0.6, 0.6, 1}},
						)
						renderer.switch_toggle(
							ui_ctx,
							&ev_ctx,
							"Enable Experimental Rendering",
							&my_switch_val,
						)
					}
				}
			}

			// ---------------------------------------------------------
			// --- OVERLAYS (Must remain at the bottom of the tree!)
			// ---------------------------------------------------------

			// 3. Modal
			if renderer.modal_begin(ui_ctx, &ev_ctx, &my_modal_open) {
				defer renderer.modal_end(ui_ctx, true)

				renderer.text(ui_ctx, &ev_ctx, "Delete User", user_style = {font_size = 24})
				renderer.text(
					ui_ctx,
					&ev_ctx,
					"Are you sure you want to permanently delete this user? This action cannot be undone.",
					user_style = {text_color = renderer.Color{0.4, 0.4, 0.4, 1}},
				)

				renderer.element_open(
					ui_ctx,
					{
						style = {
							direction = .ROW,
							gap = 12,
							justify_content = .END,
							width = lc.Percent{100},
							padding = [4]f32{16, 0, 0, 0},
						},
					},
				)
				defer renderer.element_close(ui_ctx)

				if renderer.button(ui_ctx, &ev_ctx, "Cancel", user_style = {bg_color = renderer.Color{0.9, 0.9, 0.9, 1}, text_color = renderer.Color{0.1, 0.1, 0.1, 1}}) do my_modal_open = false
				if renderer.button(
					ui_ctx,
					&ev_ctx,
					"Confirm Delete",
					user_style = {bg_color = renderer.Color{0.9, 0.2, 0.2, 1}},
				) {
					my_modal_open = false
				}
			}

			// 4. Toast Notification
			renderer.element_open(
				ui_ctx,
				{
					style = {
						position = .FIXED,
						right = 24.0,
						bottom = 24.0,
						direction = .COLUMN_REVERSE,
						z_index = 4000,
					},
				},
			)
			renderer.toast(
				ui_ctx,
				&ev_ctx,
				"Action Successful",
				"Component library integration complete.",
				.SUCCESS,
				&my_toast_open,
			)
			renderer.element_close(ui_ctx)

			// 5. Context Menu
			if renderer.context_menu_begin(
				ui_ctx,
				&ev_ctx,
				my_context_x,
				my_context_y,
				&my_context_open,
			) {
				defer renderer.context_menu_end(ui_ctx, true)
				if renderer.button(ui_ctx, &ev_ctx, "Copy ID", user_style = {text_align = .LEFT}) do my_context_open = false
				if renderer.button(ui_ctx, &ev_ctx, "Inspect Element", user_style = {text_align = .LEFT}) do my_context_open = false
				if renderer.button(ui_ctx, &ev_ctx, "Delete Node", user_style = {text_align = .LEFT, text_color = renderer.Color{0.9, 0.2, 0.2, 1}}) do my_context_open = false
			}
		}
